	if x < 0.001 then
		base *= 6 + math.random() * 4 -- titánico
	elseif x < 0.01 then
		base *= 3 + math.random() * 3 -- gigante
	elseif x < 0.1 then
		base *= 1.5 + math.random() * 1.5 -- pesado
	end
	return math.floor(base)
end

-- Números absurdos: 1.5K, 20M, 3.2B, 1T, 4Qa, 7Qi, 2Sx...
local Format = {}

local SUFFIXES = { "", "K", "M", "B", "T", "Qa", "Qi", "Sx", "Sp", "Oc", "No", "Dc", "UDc", "DDc", "TDc", "QaDc", "QiDc" }

function Format.num(n: number): string
	if n ~= n then
		return "0"
	end
	if n < 0 then
		return "-" .. Format.num(-n)
	end
	if n < 1000 then
		return tostring(math.floor(n))
	end
	local tier = math.floor(math.log10(n) / 3)
	if tier >= #SUFFIXES then
		return string.format("%.2e", n)
	end
	local scaled = n / 10 ^ (tier * 3)
	local s
	if scaled >= 100 then
		s = string.format("%d", math.floor(scaled))
	elseif scaled >= 10 then
		s = string.format("%.1f", math.floor(scaled * 10) / 10)
	else
		s = string.format("%.2f", math.floor(scaled * 100) / 100)
	end
	if string.find(s, ".", 1, true) then
		s = s:gsub("0+$", ""):gsub("%.$", "")
	end
	return s .. SUFFIXES[tier + 1]
end

function Format.time(sec: number): string
	sec = math.max(0, math.floor(sec))
	local d = math.floor(sec / 86400)
	local h = math.floor(sec % 86400 / 3600)
	local m = math.floor(sec % 3600 / 60)
	local s = sec % 60
	if d > 0 then
		return string.format("%dd %dh", d, h)
	elseif h > 0 then
		return string.format("%dh %02dm", h, m)
	end
	return string.format("%d:%02d", m, s)
end

function Format.oneIn(chance: number): string
	if chance <= 0 then
		return "—"
	end
	return "1/" .. Format.num(1 / chance)
end

function Format.hex(c: Color3): string
	return string.format("#%02X%02X%02X", math.floor(c.R * 255), math.floor(c.G * 255), math.floor(c.B * 255))
end

return Format

	local e = T[key]
	local s = e and (isSpanish and e.es or e.en) or key
	if select("#", ...) > 0 then
		local ok, res = pcall(string.format, s, ...)
		if ok then
			return res
		end
	end
	return s

	if not item then
		return "?"

end

local function mk(model, name, size, cf, color, material, shape)
	local p = Instance.new("Part")
	p.Name = name
	p.Anchored = true
	p.CanCollide = false
	p.CanQuery = false
	p.CastShadow = true
	p.Size = size
	if shape then
		p.Shape = shape
	end
	p.CFrame = cf
	p.Color = color
	p.Material = material or Enum.Material.SmoothPlastic
	p.TopSurface = Enum.SurfaceType.Smooth
	p.BottomSurface = Enum.SurfaceType.Smooth
	p.Parent = model
	return p
end

local function lookFor(god)
	return GodModel._lookFor(god)
end
function GodModel._lookFor(god)
	local base = MYTH_STYLE[god.myth] or MYTH_STYLE.greek
	local look = {}
	for k, v in pairs(base) do
		look[k] = v
	end
	for k, v in pairs(LOOKS[god.id] or {}) do
		look[k] = v
	end
	return look
end

local templates = nil
local function template(id)
	if templates == nil then
		templates = game:GetService("ReplicatedStorage"):FindFirstChild("GodTemplates") or false
	end
	return templates and templates:FindFirstChild(id) or nil
end
GodModel.template = template

-- Animación de caminar/correr: balancea brazos y piernas alrededor de sus articulaciones
local rest = setmetatable({}, { __mode = "k" })
function GodModel.animate(model, phase, amp)
	if not model:GetAttribute("Animated") then
		return
	end
	local pivot = model:GetPivot()
	local cache = rest[model]
	if not cache then
		cache = {}
		local scale = model:GetScale()
		for _, name in ipairs({ "LegL", "LegR", "ArmL", "ArmR" }) do
			local p = model:FindFirstChild(name)
			if p and p:GetAttribute("Joint") then
				cache[name] = { part = p, rest = pivot:ToObjectSpace(p.CFrame), joint = p:GetAttribute("Joint") * scale }
			end
		end
		rest[model] = cache
	end
	local s = math.sin(phase) * amp
	local angles = { LegL = s, LegR = -s, ArmL = -s * 0.8, ArmR = s * 0.5 }
	for name, c in pairs(cache) do
		local j = CFrame.new(c.joint)
		c.part.CFrame = pivot * j * CFrame.Angles(angles[name], 0, 0) * j:Inverse() * c.rest
	end
end

	return lookFor(god)
end

-- Devuelve el modelo del dios: avatar de Roblox si existe, si no, figura de piezas
function GodModel.build(god, rarityColor)
	local RS = game:GetService("ReplicatedStorage")
	local folder = RS:FindFirstChild("AvatarTemplates")
	local t = folder and folder:FindFirstChild(god.id)
	if not t and GodModel.avatarBuilder then
		t = GodModel.avatarBuilder(god)
	end
	if t then
		local m = t:Clone()
		m.Name = god.id
		return m
	end
	return GodModel.buildParts(god, rarityColor)
end

-- Canales de comunicación servidor <-> cliente
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local Net = {}

local EVENTS = {
	"State", -- servidor -> cliente: estado del jugador
	"Announce", -- servidor -> cliente: aviso en el chat / cartel
	"Toast", -- servidor -> cliente: mensaje corto
	"Reveal", -- servidor -> cliente: animación al abrir un sobre
	"Knock", -- servidor -> cliente: empujón por un golpe
	"Offer", -- servidor -> cliente: ofrecer una compra (p. ej. salvar el sobre)
	"Action", -- cliente -> servidor: comprar, equipar, AFK...
	"Combo", -- cliente -> servidor: acierto en la cinta
}

function Net.init()
	local folder = Instance.new("Folder")
	folder.Name = "Remotes"
	for _, name in ipairs(EVENTS) do
		local ev = Instance.new("RemoteEvent")
		ev.Name = name
		ev.Parent = folder
		Net[name] = ev
	end
	folder.Parent = ReplicatedStorage
end

function Net.toast(player, key, ...)
	Net.Toast:FireClient(player, key, { ... })
end

-- args: lista; cada elemento puede ser texto/número o {rarity=n}, {zone=n}, {god=id}...
function Net.announce(target, key, args, opts)
	local payload = { key = key, args = args or {}, opts = opts or {} }
	if target == "all" then
		Net.Announce:FireAllClients(payload)
	else
		Net.Announce:FireClient(target, payload)
	end
end

return Net

-- Guardado del progreso con DataStore
local DataStoreService = game:GetService("DataStoreService")

local Data = {}
Data.profiles = {} -- [player] = datos
Data.elapsed = {} -- [player] = segundos fuera del juego

local store = nil
local function getStore()
	if store == nil then
		local ok, res = pcall(function()
			return DataStoreService:GetDataStore("StealAGod_v1")
		end)
		store = ok and res or false
	end
	return store or nil
end

local function defaults()
	return {
		v = 1,
		money = 0,
		totalMoney = 0,
		speed = 0,
		bestSpeed = 0,
		playtime = 0,
		treadmill = 1,
		plot = 1,
		altars = 3,
		altarPacks = {}, -- { {r=, rem=, dawn=}, ... }
		queue = {}, -- sobres esperando altar
		gods = {}, -- [godId] = cantidad (para el índice)
		godList = nil, -- { {id=, kg=}, ... } cada dios con su peso
		coaches = {}, -- ["heracles:1"] = cantidad
		auras = {}, -- [auraId] = true
		aura = nil,
		weapon = "bat",
		settings = { sfx = true, particles = true, shake = true, othersAuras = true, ui = 2 },
		savedAt = os.time(),
	}
end

local function merge(data)
	local d = defaults()
	if type(data) ~= "table" then
		return d
	end
	for k, v in pairs(data) do
		d[k] = v
	end
	return d
end

function Data.load(player)
	local s = getStore()
	local raw = nil
	if s then
		for attempt = 1, 3 do
			local ok, res = pcall(function()
				return s:GetAsync("u_" .. player.UserId)
			end)
			if ok then
				raw = res
				break
			end
			task.wait(attempt)
		end
	end
	local d = merge(raw)
	-- versiones antiguas: convertir los dioses a la lista con kilos
	if type(d.godList) ~= "table" then
		d.godList = {}
		for id, count in pairs(d.gods) do
			for _ = 1, count do
				table.insert(d.godList, { id = id, kg = 100 })
			end
		end
	end
	-- Progreso mientras estabas fuera: los sobres siguen abriéndose
	local elapsed = math.max(0, os.time() - (d.savedAt or os.time()))
	Data.elapsed[player] = raw and elapsed or 0
	for _, pack in ipairs(d.altarPacks) do
		pack.rem = math.max(0, (pack.rem or 0) - elapsed)
	end
	Data.profiles[player] = d
	return d
end

function Data.save(player)
	local d = Data.profiles[player]
	local s = getStore()
	if not d or not s then
		return false
	end
	d.savedAt = os.time()
	local ok = pcall(function()
		s:UpdateAsync("u_" .. player.UserId, function()
			return d
		end)
	end)
	return ok
end

function Data.release(player)
	Data.save(player)
	Data.profiles[player] = nil
	Data.elapsed[player] = nil
end

return Data

	folder.Parent = ReplicatedStorage
	for id, info in pairs(Config.MODELS) do
		local ok, asset = pcall(function()
			return InsertService:LoadAsset(info.asset)
		end)
		if ok and asset then
			local m = asset:FindFirstChildOfClass("Model") or asset
			m.Name = id
			for _, p in ipairs(m:GetDescendants()) do
				if p:IsA("BasePart") then
					p.Anchored = true
					p.CanCollide = false
					p.CanQuery = false
					p.CanTouch = false
					local j = info.joints[p.Name]
					if j then
						p:SetAttribute("Joint", j)
					end
				end
			end
			local root = Instance.new("Part")
			root.Name = "Root"

			root.CanCollide = false
			root.CanQuery = false
			root.Transparency = 1
			root.Parent = m
			m.PrimaryPart = root
			m:SetAttribute("Animated", true)
			m.Parent = folder
		else
			warn("No se pudo cargar el modelo", id, asset)
		end
	end

			p.Massless = true
			p.Anchored = false
		end
	end
	hrp.Anchored = true
	-- de pie, con los pies en Y = 0
	local pivotH = hum.HipHeight + hrp.Size.Y / 2
	rig:PivotTo(CFrame.new(0, pivotH, 0))
	rig:SetAttribute("PivotH", pivotH)
	rig:SetAttribute("Avatar", true)
	if hum:FindFirstChild("Animator") == nil then
		Instance.new("Animator").Parent = hum
	end
	-- accesorios: se toman de la figura de piezas y se sueldan al avatar
	local old = G.GodModel.buildParts(god, Color3.new(1, 1, 1))
	local oldHead = old:FindFirstChild("Head") or old:FindFirstChild("FlameHead")
	local rigHead = rig:FindFirstChild("Head")
	local k = (rigHead.Size.Y / 1.2) * 0.96
	old:ScaleTo(k)
	local oldTorso = old:FindFirstChild("Torso")
	local oldArmR = old:FindFirstChild("ArmR")
	local oldHand = oldArmR.CFrame * CFrame.new(0, -oldArmR.Size.Y / 2, 0) * CFrame.Angles(math.rad(-35), 0, 0)
	local animalHead = L.head ~= nil and L.head ~= "goat"
	if animalHead then
		rigHead.Transparency = 1
		local face = rigHead:FindFirstChildOfClass("Decal")
		if face then
			face:Destroy()
		end
	end
	local acc = Instance.new("Folder")
	acc.Name = "Accessories"
	acc.Parent = rig
	local headTop = (L.head == nil or L.head == "goat") and old:FindFirstChild("Head") and old.Head.Position.Y - 0.7 * k or -math.huge
	for _, p in ipairs(old:GetChildren()) do
		if p:IsA("BasePart") then
			local name = p.Name
			local target, rel
			if HEAD[name] or (name == "Head" and animalHead) then
				if name == "Star" then
					continue
				end
				target, rel = rigHead, oldHead.CFrame:ToObjectSpace(p.CFrame)
			elseif ITEM[name] then
				target, rel = rig:FindFirstChild("RightHand"), oldHand:ToObjectSpace(p.CFrame)
			elseif TORSO[name] then
				if name == "Star" and p.Position.Y > headTop then
					target, rel = rigHead, oldHead.CFrame:ToObjectSpace(p.CFrame)
				else
					target, rel = rig:FindFirstChild("UpperTorso"), oldTorso.CFrame:ToObjectSpace(p.CFrame)
				end
			end
			if target then
				p.Anchored = false
				p.CanCollide = false
				p.CanQuery = false
				p.Massless = true
				p.CFrame = target.CFrame * rel
				p.Parent = acc
				weld(target, p)
			end
		end
	end
	old:Destroy()
	rig.PrimaryPart = hrp
	rig.Parent = folder
	return rig
end

-- Reproduce una animación en un avatar (servidor: guardianes)
function Avatars.play(rig, name, speed)
	local hum = rig:FindFirstChildOfClass("Humanoid")
	local animator = hum and hum:FindFirstChildOfClass("Animator")
	if not animator then
		return
	end
	rig:SetAttribute("Anim", name)
	local tracks = rig:GetAttribute("_t") and nil
	Avatars._tracks = Avatars._tracks or setmetatable({}, { __mode = "k" })
	local set = Avatars._tracks[rig]
	if not set then
		set = {}
		Avatars._tracks[rig] = set
	end
	if set.current == name then
		if set[name] and speed then
			set[name]:AdjustSpeed(speed)
		end
		return
	end
	if set.current and set[set.current] then
		set[set.current]:Stop(0.2)
	end
	if not set[name] then
		local a = Instance.new("Animation")
		a.AnimationId = Avatars.ANIM[name]
		set[name] = animator:LoadAnimation(a)
		set[name].Looped = true
	end
	set[name]:Play(0.2)
	if speed then
		set[name]:AdjustSpeed(speed)
	end
	set.current = name
end

function Avatars.init(g)
	G = g
	folder = Instance.new("Folder")
	folder.Name = "AvatarTemplates"
	folder.Parent = ReplicatedStorage
	G.GodModel.avatarBuilder = Avatars.make
end

return Avatars

	local god, kg = entry.god, entry.kg

	m:ScaleTo(scale)

function World.makeGuardian(i)
	local zone = G.Config.ZONES[i]
	local gd = zone.guardian
	local GMYTH = { minotaur = "greek", dullahan = "celtic", sphinx = "egyptian", jaguar = "aztec", lamassu = "babylonian", fenrir = "norse" }
	local av = G.Avatars and G.Avatars.make({ id = "g_" .. gd.kind, myth = GMYTH[gd.kind] or "greek", r = 0, name = gd.en })
	if av then
		local m = av:Clone()
		m.Name = "Guardian" .. i
		local k = 4.4 + i * 0.3
		m:ScaleTo(k)
		local h = (m:GetAttribute("PivotH") or 3) * k
		local rp = m.PrimaryPart
		local ring = part({ Name = "Scorch", Shape = Enum.PartType.Cylinder, Size = Vector3.new(0.2, 12 * k / 4, 12 * k / 4), CFrame = rp.CFrame * CFrame.new(0, -h + 0.08, 0) * CFrame.Angles(0, 0, math.rad(90)), Color = Color3.fromRGB(255, 50, 30), Material = Enum.Material.Neon, Transparency = 0.6, CanCollide = false, Parent = m })

		dec.Face = Enum.NormalId.Right

		dec.Parent = ring
		ring.Anchored = false
		local w = Instance.new("WeldConstraint")
		w.Part0 = rp
		w.Part1 = ring
		w.Parent = ring
		guardianFX(m, m:FindFirstChild("UpperTorso") or rp, k / 3, gd, i)
		m.Parent = root
		return m, h
	end
	local tmpl = G.GodModel.template(gd.kind)
	if tmpl then
		local m = tmpl:Clone()
		m.Name = "Guardian" .. i

		dec.Face = Enum.NormalId.Right
		dec.Texture = G.Config.VFX.runes
		dec.Color3 = Color3.fromRGB(255, 80, 40)
		dec.Parent = ring
		guardianFX(m, m:FindFirstChild("Body") or rp, k * 0.6, gd, i)
		m.Parent = root
		return m, 0
	end
	local s = 1.4 + (i - 1) * 0.15
	local m = Instance.new("Model")
	m.Name = "Guardian" .. i
	local rootPart = part({ Name = "Root", Size = Vector3.new(8 * s, 8 * s, 12 * s), Position = Vector3.new(0, 9 * s, 0), Color = gd.color, Material = Enum.Material.SmoothPlastic, CanCollide = false, Parent = m })
	local function add(name, size, offset, color, mat, shape)
		local p = part({ Name = name, Size = size * s, Color = color or gd.color, Material = mat or Enum.Material.SmoothPlastic, CanCollide = false, Parent = m })
		if shape then
			p.Shape = shape
		end
		p.CFrame = rootPart.CFrame * CFrame.new(offset * s)
		local w = Instance.new("WeldConstraint")
		w.Part0 = rootPart
		w.Part1 = p
		w.Parent = p
		p.Anchored = false
		return p
	end
	-- patas
	for _, o in ipairs({ Vector3.new(-3, -6, -4), Vector3.new(3, -6, -4), Vector3.new(-3, -6, 4), Vector3.new(3, -6, 4) }) do
		add("Leg", Vector3.new(2.4, 5, 2.4), o)
	end
	local headColor = gd.color
	if gd.kind == "sphinx" then
		headColor = Color3.fromRGB(230, 200, 140)
	end
	local head
	if gd.kind ~= "dullahan" then
		head = add("Head", Vector3.new(6, 6, 6), Vector3.new(0, 4, -8), headColor)
		add("EyeL", Vector3.new(1, 1, 0.4), Vector3.new(-1.5, 5, -11.1), Color3.fromRGB(255, 60, 40), Enum.Material.Neon)
		add("EyeR", Vector3.new(1, 1, 0.4), Vector3.new(1.5, 5, -11.1), Color3.fromRGB(255, 60, 40), Enum.Material.Neon)
	else
		add("Flame", Vector3.new(3, 3, 3), Vector3.new(0, 6, -6), Color3.fromRGB(120, 255, 120), Enum.Material.Neon, Enum.PartType.Ball)
	end
	if gd.kind == "minotaur" then
		add("HornL", Vector3.new(1, 1, 4), Vector3.new(-3.5, 7.5, -8), Color3.fromRGB(240, 230, 210))
		add("HornR", Vector3.new(1, 1, 4), Vector3.new(3.5, 7.5, -8), Color3.fromRGB(240, 230, 210))
	elseif gd.kind == "sphinx" then
		add("Nemes", Vector3.new(7, 3, 6.5), Vector3.new(0, 7, -7.5), Color3.fromRGB(40, 70, 170))
	elseif gd.kind == "jaguar" then
		for k = 1, 6 do
			add("Spot", Vector3.new(1.4, 0.3, 1.4), Vector3.new(-2.5 + (k % 3) * 2.5, 4.1, -3 + math.floor(k / 3) * 3), Color3.fromRGB(40, 25, 15))
		end
	elseif gd.kind == "lamassu" then
		add("WingL", Vector3.new(0.5, 6, 12), Vector3.new(-4.5, 5, 0), Color3.fromRGB(220, 200, 150))
		add("WingR", Vector3.new(0.5, 6, 12), Vector3.new(4.5, 5, 0), Color3.fromRGB(220, 200, 150))
		add("Beard", Vector3.new(4, 3, 1), Vector3.new(0, 1.5, -11), Color3.fromRGB(60, 40, 30))
	elseif gd.kind == "fenrir" then
		add("EarL", Vector3.new(1.5, 3, 1), Vector3.new(-2, 8, -7), gd.color)
		add("EarR", Vector3.new(1.5, 3, 1), Vector3.new(2, 8, -7), gd.color)
		add("Chain", Vector3.new(8.5, 1, 1), Vector3.new(0, 1, -5), Color3.fromRGB(180, 180, 190), Enum.Material.Metal)
	end
	add("Tail", Vector3.new(1, 1, 6), Vector3.new(0, 2, 9))
	m.PrimaryPart = rootPart
	-- contorno amenazante y materiales
	for _, p in ipairs(m:GetDescendants()) do
		if p:IsA("BasePart") and p.Material == Enum.Material.SmoothPlastic then
			p.Material = (gd.kind == "fenrir" or gd.kind == "jaguar") and Enum.Material.Fabric or Enum.Material.Slate
		end
	end
	local hl = Instance.new("Highlight")
	hl.OutlineColor = Color3.fromRGB(255, 60, 50)
	hl.FillColor = Color3.fromRGB(255, 40, 40)
	hl.FillTransparency = 0.9
	hl.DepthMode = Enum.HighlightDepthMode.Occluded
	hl.Parent = m
	local flames = Instance.new("ParticleEmitter")
	flames.Rate = 45
	flames.Lifetime = NumberRange.new(0.5, 1)
	flames.Speed = NumberRange.new(2, 6)
	flames.Acceleration = Vector3.new(0, 10, 0)
	flames.SpreadAngle = Vector2.new(60, 60)
	flames.LightEmission = 1
	flames.Color = ColorSequence.new(Color3.fromRGB(255, 120, 40), Color3.fromRGB(180, 20, 20))
	flames.Size = NumberSequence.new({ NumberSequenceKeypoint.new(0, 2.5 * s), NumberSequenceKeypoint.new(1, 0) })
	flames.Transparency = NumberSequence.new(0.2, 1)
	flames.Parent = rootPart
	local scorch = part({ Name = "Scorch", Shape = Enum.PartType.Cylinder, Size = Vector3.new(0.2, 16 * s, 16 * s), Color = Color3.fromRGB(255, 40, 30), Material = Enum.Material.Neon, Transparency = 0.6, CanCollide = false, Parent = m })

	sw.Part1 = scorch
	sw.Parent = scorch

	smoke.Rate = 6
	smoke.Lifetime = NumberRange.new(1.5, 2.5)
	smoke.Speed = NumberRange.new(1, 3)
	smoke.Color = ColorSequence.new(Color3.fromRGB(80, 20, 20))
	smoke.Transparency = NumberSequence.new(0.6, 1)
	smoke.Size = NumberSequence.new(2, 5)
	smoke.Parent = rootPart

	gbb:SetAttribute("Guardian", i)

	return m, 9 * s
end

local function buildCorridor()
	local C = G.Config
	local L = C.ZONE_LENGTH
	local W = C.CORRIDOR_HALF_WIDTH
	for i, zone in ipairs(C.ZONES) do
		local z0 = World.zoneStart(i)
		local zp = {}
		zp.floor = part({ Name = "Zone" .. i, Size = Vector3.new(W * 2 + 260, 4, L), Position = Vector3.new(0, -2, z0 + L / 2), Color = zone.ground, Material = Enum.Material.SmoothPlastic })
		local tex = Instance.new("Texture")
		tex.Texture = C.VFX.studs
		tex.Face = Enum.NormalId.Top
		tex.StudsPerTileU = 4
		tex.StudsPerTileV = 4
		tex.Transparency = 0.15
		tex.Parent = zp.floor
		-- pista central más clara
		part({ Name = "Runway", Size = Vector3.new(W * 2, 0.2, L), Position = Vector3.new(0, 0.1, z0 + L / 2), Color = zone.ground:Lerp(Color3.new(1, 1, 1), 0.18), Material = Enum.Material.SmoothPlastic })
		-- muros: bajos y de color, con franja de neón; colisión invisible alta
		for _, k in ipairs({ -1, 1 }) do
			part({ Name = "Wall", Size = Vector3.new(2, 80, L), Position = Vector3.new(k * (W + 3), 40, z0 + L / 2), Transparency = 1 })
			part({ Name = "WallVisible", Size = Vector3.new(4, 9, L), Position = Vector3.new(k * (W + 3), 4.5, z0 + L / 2), Color = zone.accent, Material = Enum.Material.SmoothPlastic })
			part({ Name = "WallNeon", Size = Vector3.new(4.4, 0.8, L), Position = Vector3.new(k * (W + 3), 9.2, z0 + L / 2), Color = zone.ground:Lerp(Color3.new(1, 1, 1), 0.35), Material = Enum.Material.Neon })
		end
		-- arco de neón en la entrada de la zona
		local archColor = zone.ground:Lerp(Color3.new(1, 1, 1), 0.3)
		for _, k in ipairs({ -1, 1 }) do
			part({ Name = "ArchPillar", Size = Vector3.new(6, 42, 6), Position = Vector3.new(k * (W + 3), 21, z0), Color = zone.accent, Material = Enum.Material.Marble })
			part({ Name = "ArchNeon", Size = Vector3.new(1.2, 42, 6.6), Position = Vector3.new(k * (W + 3) - k * 3.2, 21, z0), Color = archColor, Material = Enum.Material.Neon })
		end
		part({ Name = "ArchNeonTop", Size = Vector3.new(W * 2 + 12, 1.2, 6.6), Position = Vector3.new(0, 41, z0), Color = archColor, Material = Enum.Material.Neon })
		local top = part({ Name = "ArchTop", Size = Vector3.new(W * 2 + 12, 6, 6), Position = Vector3.new(0, 44.5, z0), Color = zone.accent, Material = Enum.Material.Marble })
		zp.sign = billboard(top, "", Color3.new(1, 1, 1), UDim2.fromOffset(520, 110), Vector3.new(0, 8, 0))
		zp.sign.MaxDistance = 140
		zp.sign:SetAttribute("Zone", i)
		CollectionService:AddTag(zp.sign, "Loc")
		if i > 1 then
			zp.curtain = part({ Name = "Curtain", Size = Vector3.new(W * 2, 40, 0.6), Position = Vector3.new(0, 20, z0), Color = archColor, Material = Enum.Material.ForceField, Transparency = 0.55, CanCollide = false })
		end
		-- puntos donde aparecen los sobres
		zp.spots = {}
		for row = 0, 1 do
			for col = -1, 1 do
				table.insert(zp.spots, Vector3.new(col * 42, 2.5, z0 + 65 + row * 70))
			end
		end
		zp.guardianHome = Vector3.new(0, 0, z0 + L - 25)
		decorZone(i, z0, zone)
		World.zoneParts[i] = zp
	end
	-- fin del pasillo (más zonas próximamente)
	local zEnd = World.zoneStart(#C.ZONES + 1)
	local wall = part({ Name = "EndWall", Size = Vector3.new(W * 2 + 4, 60, 4), Position = Vector3.new(0, 30, zEnd), Color = Color3.fromRGB(20, 15, 35), Material = Enum.Material.ForceField })
	World.endSign = billboard(wall, "", Color3.fromRGB(180, 160, 255), UDim2.fromOffset(520, 110), Vector3.new(0, 0, 0))
	World.endSign.MaxDistance = 140
	World.endSign:SetAttribute("EndSign", true)
	CollectionService:AddTag(World.endSign, "Loc")
	-- niebla de la noche en la entrada del pasillo
	World.nightGate = part({ Name = "NightGate", Size = Vector3.new(W * 2 + 4, 40, 2), Position = Vector3.new(0, 20, C.CORRIDOR_START - 4), Color = Color3.fromRGB(30, 10, 60), Material = Enum.Material.ForceField, Transparency = 1, CanCollide = false })
end

local function uniqueGods(d)
	local n = 0
	for _, c in pairs(d.gods) do
		if c > 0 then
			n += 1
		end
	end
	return n
end
Core.uniqueGods = uniqueGods

local function mythComplete(d, myth)
	for _, g in ipairs(C.GODS) do
		if g.myth == myth and (d.gods[g.id] or 0) <= 0 then
			return false
		end
	end
	return true
end
Core.mythComplete = mythComplete

function Core.mults(player)
	local d = G.Data.profiles[player]
	local s = Core.sessions[player]
	if not d or not s then
		return 1, 1, 1
	end
	local speedAdd, moneyAdd, speedMul, moneyMul = 0, 0, 1, 1
	local n = uniqueGods(d)
	for _, rw in ipairs(C.INDEX_REWARDS) do
		if n >= rw.count then
			speedAdd += rw.speed or 0
			moneyAdd += rw.money or 0
			speedMul *= rw.speedMult or 1
			moneyMul *= rw.moneyMult or 1
		end
	end
	for key, myth in pairs(C.MYTHS) do
		if mythComplete(d, key) then
			speedAdd += myth.bonus
			moneyAdd += myth.bonus
		end
	end
	-- entrenadores: los 3 mejores
	local list = {}
	for key, count in pairs(d.coaches) do
		if count > 0 then
			local id, tier = key:match("^(%w+):(%d+)$")
			tier = tonumber(tier)
			for _, co in ipairs(C.COACHES) do
				if co.id == id and tier and C.COACH_TIERS[tier] then
					for _ = 1, count do
						table.insert(list, co.bonus * C.COACH_TIERS[tier].mult)
					end
				end
			end
		end
	end
	table.sort(list, function(a, b)
		return a > b
	end)
	for k = 1, math.min(C.COACH_EQUIP, #list) do
		speedAdd += list[k]
	end
	local aura = d.aura and C.AURA_BY_ID[d.aura]
	if aura then
		speedAdd += aura.speed or 0
		moneyAdd += aura.money or 0
	end
	if s.passes.vip then
		speedAdd += 0.1
		moneyAdd += 0.1
	end
	if s.passes.speed2x then
		speedMul *= 2
	end
	if s.passes.money2x then
		moneyMul *= 2
	end
	if os.clock() < (s.speedBoostUntil or 0) then
		speedMul *= 3
	end
	local luck = 1
	if s.passes.luck2x then
		luck *= 2
	end
	if os.clock() < (G.serverLuckUntil or 0) then
		luck *= 3
	end
	return (1 + speedAdd) * speedMul, (1 + moneyAdd) * moneyMul, luck
end

function Core.pedestalCount(player)
	local d = G.Data.profiles[player]
	local s = Core.sessions[player]
	if not d then
		return 0
	end
	return C.pedestalsFor(d.plot) + ((s and s.passes.moreSlots) and 5 or 0)
end

function Core.altarCount(player)
	local d = G.Data.profiles[player]
	local s = Core.sessions[player]
	if not d then
		return 0
	end
	return d.altars + ((s and s.passes.moreSlots) and 2 or 0)
end

-- Los mejores dioses (por ingresos, contando los kilos) que caben en la parcela
function Core.placedGods(player)
	local d = G.Data.profiles[player]
	local all = {}
	for _, e in ipairs(d.godList or {}) do
		local g = C.GOD_BY_ID[e.id]
		if g then
			table.insert(all, { god = g, kg = e.kg or 100, value = C.RARITIES[g.r].income * C.kgMult(e.kg) })
		end
	end
	table.sort(all, function(x, y)
		if x.value ~= y.value then
			return x.value > y.value
		end
		return x.god.index < y.god.index
	end)
	local n = Core.pedestalCount(player)
	local placed = {}
	for i = 1, math.min(n, #all) do
		placed[i] = all[i]
	end
	return placed
end

function Core.income(player)
	local _, moneyMult = Core.mults(player)
	local sum = 0
	for _, e in ipairs(Core.placedGods(player)) do
		sum += e.value
	end
	return sum * moneyMult
end

function Core.refreshPlot(player)
	local s = Core.sessions[player]
	local d = G.Data.profiles[player]
	if not s or not d or not s.plot then
		return
	end
	local plot = s.plot
	local count = Core.pedestalCount(player)
	G.World.pedestalVisible(plot, count)
	G.World.altarVisible(plot, Core.altarCount(player))
	local placed = Core.placedGods(player)
	for i = 1, #plot.pedestals do
		G.World.setStatue(plot, i, i <= count and placed[i] or nil)
	end
	if s.plotStyled ~= d.plot then
		s.plotStyled = d.plot
		G.World.styleplot(plot, d.plot)
	end
	plot.signGui.Label.Text = string.format("<b>%s</b>\n%s %d", player.DisplayName, "⭐", d.plot)
end

function Core.bestWeapon(d)
	local best, order = "bat", 0
	for key, myth in pairs(C.MYTHS) do
		local w = C.WEAPONS[myth.weapon]
		if w and w.order > order and mythComplete(d, key) then
			best, order = myth.weapon, w.order
		end
	end
	return best
end

function Core.addGod(player, god, kg)
	local d = G.Data.profiles[player]
	local isNew = (d.gods[god.id] or 0) == 0
	d.gods[god.id] = (d.gods[god.id] or 0) + 1
	table.insert(d.godList, { id = god.id, kg = kg or 100 })
	if isNew then
		local w = Core.bestWeapon(d)
		if w ~= d.weapon then
			d.weapon = w
			for key, myth in pairs(C.MYTHS) do
				if myth.weapon == w then
					G.Net.announce(player, "weaponUnlocked", { { myth = key }, { weapon = w } }, { color = "#FFD23F", big = true })
				end
			end
			G.Combat.giveWeapon(player)
		end
	end
	Core.refreshPlot(player)
	return isNew
end

	if d.altars >= C.ALTARS_MAX then

function actions.buyCoach(player, id)
	local d = G.Data.profiles[player]
	for _, co in ipairs(C.COACHES) do
		if co.id == id then
			if Core.spend(player, co.cost) then
				local key = id .. ":1"
				d.coaches[key] = (d.coaches[key] or 0) + 1
				G.Net.toast(player, "bought")
			end
			return
		end
	end
end

function actions.fuseCoach(player, id, tier)
	local d = G.Data.profiles[player]
	tier = tonumber(tier)
	if type(id) ~= "string" or not tier or tier >= #C.COACH_TIERS then
		return
	end
	local key = id .. ":" .. tier
	if (d.coaches[key] or 0) >= 2 then
		d.coaches[key] -= 2
		local up = id .. ":" .. (tier + 1)
		d.coaches[up] = (d.coaches[up] or 0) + 1
		G.Net.toast(player, "bought")
	end
end

function actions.buyAura(player, id)
	local d = G.Data.profiles[player]
	local a = C.AURA_BY_ID[id]
	if not a or d.auras[id] then
		return
	end
	if Core.spend(player, a.cost) then
		d.auras[id] = true
		d.aura = id
		Core.applyAura(player)
		G.Net.toast(player, "bought")
	end
end

	local d = G.Data.profiles[player]
	if d.auras[id] then
		d.aura = id
		Core.applyAura(player)
	end
end

local function onTreadmill(s, hrp)
	if not s.plot or not hrp then
		return false
	end
	local t = s.plot.treadmill
	local rel = t.CFrame:PointToObjectSpace(hrp.Position)
	return math.abs(rel.X) <= t.Size.X / 2 + 0.5 and math.abs(rel.Z) <= t.Size.Z / 2 + 0.5 and rel.Y > 0 and rel.Y < 8
end

local function freePlot()
	for _, plot in ipairs(G.World.plots) do

local function setupLeaderstats(player)
	local ls = Instance.new("Folder")
	ls.Name = "leaderstats"
	for _, n in ipairs({ "⚡", "💰", "⏱️" }) do
		local v = Instance.new("StringValue")
		v.Name = n
		v.Value = "0"
		v.Parent = ls
	end
	ls.Parent = player
end

local function onCharacter(player, char)
	local s = Core.sessions[player]
	local d = G.Data.profiles[player]
	if not s or not d then
		return
	end
	local hrp = char:WaitForChild("HumanoidRootPart", 10)
	local hum = char:WaitForChild("Humanoid", 10)
	if not hrp or not hum then
		return
	end
	if s.plot then
		task.wait(0.1)
		hrp.CFrame = s.plot.spawnCF
	end
	hum.WalkSpeed = C.walkSpeedFor(d.speed)
	hum.Died:Connect(function()
		if s.carrying then
			G.Packs.dropCarried(player, false)
		end
	end)
	Core.applyAura(player)
	overhead(player, char)
	G.Combat.giveWeapon(player)
	if s.afk then
		task.wait(0.5)
		Core.setAfk(player, true)
	end
end

local function onJoin(player)
	local d = G.Data.load(player)
	if not player.Parent then
		return
	end
	local s = {
		passes = {},
		combo = 1,
		lastTap = 0,
		lastHit = 0,
		invulnUntil = 0,
		shieldUntil = 0,
		speedBoostUntil = 0,
		afk = false,
		joinedAt = os.clock(),
		carrying = nil,
	}
	Core.sessions[player] = s
	G.Shop.checkPasses(player)
	local plot = freePlot()
	if plot then
		plot.owner = player
		s.plot = plot
		plot.folder:SetAttribute("Owner", player.UserId)
	end
	setupLeaderstats(player)
	task.spawn(function()
		for id in pairs(d.gods) do
			local g = C.GOD_BY_ID[id]
			if g then
				G.Avatars.make(g)
			end
		end
	end)
	Core.refreshPlot(player)
	-- dinero ganado mientras no estabas (50 %, máximo 8 h)
	local away = G.Data.elapsed[player] or 0
	if away >= 60 then
		local amount = math.min(away, 8 * 3600) * Core.income(player) * 0.5
		if amount > 0 then
			s.offline = amount
			task.delay(3, function()
				if player.Parent then
					G.Net.Offer:FireClient(player, "offline", amount, away)
				end
			end)
		end
	end
	-- volver del "rejoin" AFK
	local jd = player:GetJoinData()
	if jd and type(jd.TeleportData) == "table" and jd.TeleportData.afk then
		s.afk = true
	end
	player.CharacterAdded:Connect(function(char)
		onCharacter(player, char)
	end)
	if player.Character then
		task.spawn(onCharacter, player, player.Character)
	end
end

local function onLeave(player)
	local s = Core.sessions[player]
	if s then
		if s.carrying then
			G.Packs.dropCarried(player, false)
		end
		if s.plot then
			for i = 1, #s.plot.pedestals do
				G.World.setStatue(s.plot, i, nil)
			end
			G.World.pedestalVisible(s.plot, 0)
			G.World.altarVisible(s.plot, 0)
			for _, a in ipairs(s.plot.altars) do
				a.prompt.Enabled = false
				a.glow.Transparency = 1
				a.gui.Label.Text = ""
			end
			s.plot.signGui.Label.Text = ""
			s.plot.treadmillGui.Label.Text = ""
			s.plot.owner = nil
			s.plot.folder:SetAttribute("Owner", nil)
		end
	end
	G.Data.release(player)
	Core.sessions[player] = nil
end

function Core.sendState(player)
	local d = G.Data.profiles[player]
	local s = Core.sessions[player]
	if not d or not s then
		return
	end
	local speedMult, moneyMult, luck = Core.mults(player)
	local altars = {}
	for i, p in ipairs(d.altarPacks) do
		altars[i] = { r = p.r, rem = p.rem, dawn = p.dawn }
	end
	local chase = nil
	if s.carrying then
		local hrp = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
		for _, guard in ipairs(G.World.guardians) do
			if guard.target == player and hrp then
				chase = math.floor((guard.model:GetPivot().Position - hrp.Position).Magnitude)
			end
		end
	end
	G.Net.State:FireClient(player, {
		chase = chase,
		settings = d.settings,
		offline = s.offline,
		money = d.money,
		speed = d.speed,
		income = Core.income(player),
		combo = s.combo,
		treadmill = d.treadmill,
		treadmillGain = C.treadmillGain(d.treadmill) * speedMult,
		plot = d.plot,
		pedestals = Core.pedestalCount(player),
		altarsMax = Core.altarCount(player),
		altarsBought = d.altars,
		altarPacks = altars,
		queue = #d.queue,
		gods = d.gods,
		coaches = d.coaches,
		auras = d.auras,
		aura = d.aura,
		weapon = d.weapon,
		passes = s.passes,
		carrying = s.carrying and s.carrying.rarity or nil,
		afk = s.afk,
		night = G.DayNight.isNight(),
		cycleLeft = G.DayNight.timeLeft(),
		speedMult = speedMult,
		moneyMult = moneyMult,
		luck = luck,
		onTreadmill = s.onTreadmill,
		zone = s.zone or 0,
		shieldLeft = math.max(0, s.shieldUntil - os.clock()),
		boostLeft = math.max(0, s.speedBoostUntil - os.clock()),
		protectedLeft = math.max(0, C.NEWBIE_PROTECTION - d.playtime),
	})
end

---------------------------------------------------------------------
-- Bucle principal
---------------------------------------------------------------------
local function tick(dt)
	local night = G.DayNight.isNight()
	for player, s in pairs(Core.sessions) do
		local d = G.Data.profiles[player]
		if d then
			local char = player.Character
			local hrp = char and char:FindFirstChild("HumanoidRootPart")
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			s.onTreadmill = onTreadmill(s, hrp)
			if os.clock() - s.lastTap > C.COMBO_DECAY then
				s.combo = 1
			end
			local speedMult = Core.mults(player)
			local gaining = s.onTreadmill or s.afk or s.passes.autoTreadmill
			if gaining then
				local combo = s.onTreadmill and not s.afk and s.combo or 1
				local gain = C.treadmillGain(d.treadmill) * combo * speedMult * (night and C.NIGHT_TREADMILL_MULT or 1)
				d.speed += gain * dt
				if d.speed > d.bestSpeed then
					d.bestSpeed = d.speed
				end
			end
			if hum then
				hum.WalkSpeed = C.walkSpeedFor(d.speed)
			end
			if hrp then
				s.zone = G.World.zoneAt(hrp.Position.Z)
			end
		end
	end
end

local acc, secAcc = 0, 0
function Core.start()
	RunService.Heartbeat:Connect(function(dt)
		tick(dt)
		acc += dt
		secAcc += dt
		if secAcc >= 1 then
			secAcc -= 1
			local night = G.DayNight.isNight()
			for player, s in pairs(Core.sessions) do
				local d = G.Data.profiles[player]
				if d then
					d.playtime += 1
					Core.addMoney(player, Core.income(player))
					G.Packs.tickAltars(player, night and C.NIGHT_OPEN_MULT or 1)
					local ls = player:FindFirstChild("leaderstats")
					if ls then
						ls["⚡"].Value = G.Format.num(d.speed)
						ls["💰"].Value = G.Format.num(d.money)
						ls["⏱️"].Value = G.Format.time(d.playtime)
					end
					local head = player.Character and player.Character:FindFirstChild("Head")
					local bb = head and head:FindFirstChild("Overhead")
					if bb then
						bb:SetAttribute("Title", titleIndex(d.bestSpeed))
						bb:SetAttribute("Income", Core.income(player))
					end
					if s.plot then
						s.plot.treadmillGui.Label.Text = string.format("⚡ +%s/s", G.Format.num(C.treadmillGain(d.treadmill) * Core.mults(player)))
					end
				end
			end
		end
		if acc >= 0.5 then
			acc = 0
			for player in pairs(Core.sessions) do
				Core.sendState(player)
			end
		end
	end)
	-- guardado automático
	task.spawn(function()
		while true do
			task.wait(120)
			for player in pairs(Core.sessions) do
				task.spawn(G.Data.save, player)
			end
		end
	end)
end

function Core.init(g)
	G = g
	C = g.Config
	G.Net.Action.OnServerEvent:Connect(function(player, name, a, b)
		local fn = type(name) == "string" and actions[name]
		if fn and G.Data.profiles[player] and Core.sessions[player] then
			fn(player, a, b)
			Core.sendState(player)
		end
	end)
	G.Net.Combo.OnServerEvent:Connect(function(player)
		local s = Core.sessions[player]
		if not s or not s.onTreadmill or s.afk then
			return
		end
		local now = os.clock()
		if now - s.lastTap < 0.3 then
			return
		end
		s.lastTap = now
		s.combo = math.min(C.COMBO_MAX, s.combo + 0.5)
	end)
	Players.PlayerAdded:Connect(onJoin)
	Players.PlayerRemoving:Connect(onLeave)
	for _, p in ipairs(Players:GetPlayers()) do
		task.spawn(onJoin, p)
	end
	game:BindToClose(function()
		for player in pairs(Core.sessions) do
			G.Data.save(player)
		end
	end)
end

return Core

-- Tirada de rareza
---------------------------------------------------------------------
local function rollRarity(zoneIndex, luck)
	local zone = C.ZONES[zoneIndex]
	-- rarezas "imposibles" (zona 3 en adelante)
	if zoneIndex >= 3 then
		for k = #C.ULTRA, 1, -1 do
			local u = C.ULTRA[k]
			if math.random() < (luck * zoneIndex / 6) / u.oneIn then
				return u.r
			end
		end
	end
	local total = 0
	local weights = {}
	local maxR = 0
	for r in pairs(zone.weights) do
		maxR = math.max(maxR, r)
	end
	for r, w in pairs(zone.weights) do
		-- la suerte potencia las dos rarezas más altas de la zona
		local ww = (r >= maxR - 1) and w * luck or w
		weights[r] = ww
		total += ww
	end
	local x = math.random() * total
	for r = 1, #C.RARITIES do
		local w = weights[r]
		if w then
			x -= w
			if x <= 0 then
				return r
			end
		end
	end
	return maxR
end
Packs.rollRarity = rollRarity

-- Probabilidades de cada zona (para mostrarlas en el juego)
function Packs.odds(zoneIndex)
	local zone = C.ZONES[zoneIndex]
	local total = 0
	for _, w in pairs(zone.weights) do
		total += w
	end
	local out = {}
	for r, w in pairs(zone.weights) do
		out[r] = w / total
	end
	if zoneIndex >= 3 then
		for _, u in ipairs(C.ULTRA) do
			out[u.r] = (zoneIndex / 6) / u.oneIn
		end
	end
	return out
end

local function makePackPart(pack)

	p.Color = rarity.color
	p.Material = pack.rarity >= 5 and Enum.Material.Neon or Enum.Material.SmoothPlastic
	local border = Instance.new("SelectionBox")
	border.Adornee = p
	border.LineThickness = 0.08
	border.Color3 = Color3.fromRGB(255, 220, 120)
	border.Parent = p
	local light = Instance.new("PointLight")
	light.Color = rarity.color
	light.Range = 10 + pack.rarity
	light.Brightness = 1 + pack.rarity * 0.2
	light.Parent = p
	if rarity.sparkle or rarity.rainbow or pack.dawn then
		local pe = Instance.new("ParticleEmitter")
		pe.Rate = 15
		pe.Lifetime = NumberRange.new(0.8, 1.6)
		pe.Speed = NumberRange.new(1, 4)
		pe.LightEmission = 1
		pe.Color = ColorSequence.new(rarity.color, Color3.new(1, 1, 1))

	end

	for _, face in ipairs({ Enum.NormalId.Front, Enum.NormalId.Back }) do
		local sg = Instance.new("SurfaceGui")
		sg.Face = face
		sg.CanvasSize = Vector2.new(300, 400)

		q.BackgroundTransparency = 1

		q.Parent = bg

		w.BackgroundTransparency = 1
		w.Size = UDim2.new(1, 0, 0.16, 0)
		w.Position = UDim2.fromScale(0, 0.74)

		w.Text = pack.kg .. " KG"
		w.TextColor3 = Color3.fromRGB(255, 230, 140)
		w.Parent = bg

	gui:SetAttribute("Rarity", pack.rarity)
	gui:SetAttribute("Kg", pack.kg)
	gui:SetAttribute("Dawn", pack.dawn == true)

		Packs.grab(player, pack)

local function placeOnGround(pack, pos)
	local y = pack.part.Size.Y / 2 + 1.4 + math.sin(os.clock() * 2 + pack.id) * 0.35
	pack.part.CFrame = CFrame.new(pos.X, y, pos.Z) * CFrame.Angles(0, math.rad(os.clock() * 50 % 360), 0)
	if pack.beam then
		pack.beam.CFrame = CFrame.new(pos + Vector3.new(0, 150, 0)) * CFrame.Angles(0, 0, math.rad(90))
	end
end

local function destroyPack(pack)
	pack.state = "gone"
	if pack.part then
		pack.part:Destroy()
	end
	if pack.beam then
		pack.beam:Destroy()
	end
	Packs.list[pack.id] = nil
end

local function spawnPack(zoneIndex, spotIndex, dawn)
	local luck = (G.serverLuckUntil and os.clock() < G.serverLuckUntil) and 3 or 1
	if dawn then
		luck *= C.DAWN_LUCK
	end
	nextId += 1
	local pack = {
		id = nextId,
		zone = zoneIndex,
		spot = spotIndex,
		rarity = rollRarity(zoneIndex, luck),
		kg = C.rollKg(),
		dawn = dawn,
		state = "ground",
		carrier = nil,
	}
	pack.home = G.World.zoneParts[zoneIndex].spots[spotIndex]
	makePackPart(pack)
	placeOnGround(pack, pack.home)
	Packs.list[pack.id] = pack
	return pack
end

-- Amanecer: se renuevan todos los sobres y se avisa en el chat
---------------------------------------------------------------------
function Packs.refreshAll()
	for _, pack in pairs(Packs.list) do
		if pack.state ~= "carried" and pack.state ~= "pending" then
			destroyPack(pack)
		end
	end
	G.Net.announce("all", "dawn", {}, { color = "#FFB347" })
	local best = {}
	for z = 1, #C.ZONES do
		local dawnSpot = math.random(1, C.PACKS_PER_ZONE)
		for i = 1, C.PACKS_PER_ZONE do
			local pack = spawnPack(z, i, i == dawnSpot)
			if pack.rarity >= C.ANNOUNCE_CHAT_FROM then
				table.insert(best, pack)
			end
		end
	end
	table.sort(best, function(a, b)
		return a.rarity > b.rarity
	end)
	for _, pack in ipairs(best) do
		local big = pack.rarity >= C.ANNOUNCE_BIG_FROM
		local color = G.Format.hex(C.RARITIES[pack.rarity].color)
		G.Net.announce("all", big and "packSpawnedBig" or "packSpawned", { { rarity = pack.rarity }, { zone = pack.zone }, pack.zone }, { color = color, big = big })
		if pack.rarity >= C.ANNOUNCE_GLOBAL_FROM then
			local host = game:GetService("Players"):GetPlayers()[1]
			pcall(function()
				MessagingService:PublishAsync("GlobalAnnounce", { key = "packGlobal", args = { { rarity = pack.rarity }, host and host.DisplayName or "?" }, color = color, jobId = game.JobId })
			end)
		end
	end
end

-- Al anochecer, los sobres que alguien llevaba se pierden
function Packs.nightfall()
	for player, s in pairs(G.Core.sessions) do
		if s.carrying then
			Packs.dropCarried(player, false)
		end
	end
end

function Packs.grab(player, pack)
	local s = G.Core.sessions[player]
	local d = G.Data.profiles[player]
	if not s or not d or pack.state == "carried" or pack.state == "gone" or pack.state == "pending" then
		return
	end
	if s.carrying or s.afk or G.DayNight.isNight() then
		return
	end
	local char = player.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not hrp or (hrp.Position - pack.part.Position).Magnitude > 14 then
		return
	end
	if d.speed < C.ZONES[pack.zone].req then
		return
	end
	pack.state = "carried"
	pack.carrier = player
	s.carrying = pack
	pack.prompt.Enabled = false
	Packs.burst(pack.part.Position, C.RARITIES[pack.rarity].color, 40 + pack.rarity * 8)
	-- aura mientras lo llevas
	if not pack.part:FindFirstChild("CarryFX") then
		local att = Instance.new("Attachment")
		att.Name = "CarryFX"
		att.Parent = pack.part
		local pe = Instance.new("ParticleEmitter")
		pe.Rate = 35
		pe.Lifetime = NumberRange.new(0.5, 1)
		pe.Speed = NumberRange.new(2, 6)
		pe.SpreadAngle = Vector2.new(180, 180)
		pe.LightEmission = 1
		pe.Color = ColorSequence.new(C.RARITIES[pack.rarity].color, Color3.new(1, 1, 1))
		pe.Size = NumberSequence.new({ NumberSequenceKeypoint.new(0, 0.9), NumberSequenceKeypoint.new(1, 0) })
		pe.Texture = C.VFX.sparkle
		pe.RotSpeed = NumberRange.new(-120, 120)
		pe.Parent = att
	end
	if pack.beam then
		pack.beam.Transparency = 0.7
	end
	-- el guardián de la zona de origen despierta y persigue
	local guard = G.World.guardians[pack.zone]
	guard.target = player
	guard.pack = pack
	if pack.rarity >= C.ANNOUNCE_CHAT_FROM then
		G.Net.announce("all", "stoleBig", { player.DisplayName, { rarity = pack.rarity }, { guardian = pack.zone } }, { color = G.Format.hex(C.RARITIES[pack.rarity].color) })
	end
end

-- Suelta el sobre que lleva el jugador. loose=true lo deja en el suelo (golpe); false lo devuelve a su sitio
function Packs.dropCarried(player, loose, at)
	local s = G.Core.sessions[player]
	if not s or not s.carrying then
		return nil
	end
	local pack = s.carrying
	s.carrying = nil
	pack.carrier = nil
	for _, guard in ipairs(G.World.guardians) do
		if guard.target == player then
			guard.target = nil
		end
	end
	if G.DayNight.isNight() then
		destroyPack(pack)
		return nil
	end
	pack.state = loose and "loose" or "ground"
	pack.prompt.Enabled = true
	local fx = pack.part:FindFirstChild("CarryFX")
	if fx then
		fx:Destroy()
	end
	if pack.beam then
		pack.beam.Transparency = 0.4
	end
	local pos = loose and at and Vector3.new(at.X, 2.5, at.Z) or pack.home
	pack.loosePos = pos
	placeOnGround(pack, pos)
	return pack
end

local function deliver(player, pack)
	local d = G.Data.profiles[player]
	local s = G.Core.sessions[player]
	local item = { r = pack.rarity, rem = C.RARITIES[pack.rarity].open, dawn = pack.dawn == true, kg = pack.kg }
	if #d.altarPacks < G.Core.altarCount(player) then
		table.insert(d.altarPacks, item)
	elseif #d.queue < C.PACK_QUEUE_MAX then
		table.insert(d.queue, item)
		G.Net.toast(player, "altarsFull")
	else
		G.Net.toast(player, "queueFull")
		return false
	end
	s.carrying = nil
	for _, guard in ipairs(G.World.guardians) do
		if guard.target == player then
			guard.target = nil
		end
	end
	if pack.rarity >= C.ANNOUNCE_CHAT_FROM then
		G.Net.announce("all", "escaped", { player.DisplayName, { rarity = pack.rarity } }, { color = G.Format.hex(C.RARITIES[pack.rarity].color) })
	end
	destroyPack(pack)
	return true
end

local function caught(player, guardIndex)
	local s = G.Core.sessions[player]
	if not s or not s.carrying then
		return
	end
	local pack = s.carrying
	G.Net.announce("all", "caught", { { guardian = guardIndex }, player.DisplayName }, { color = "#FF5A5A" })
	local product = C.PRODUCTS.savePack
	if product.id ~= 0 then
		-- ofrecer "Salvar el sobre" durante 10 s
		s.carrying = nil
		pack.carrier = nil
		pack.state = "pending"
		pack.part.Transparency = 1
		pendingSave[player] = { pack = pack, untilT = os.clock() + 10 }
		G.Net.Offer:FireClient(player, "savePack")
		task.delay(10.5, function()
			local p = pendingSave[player]
			if p and p.pack == pack then
				pendingSave[player] = nil
				if pack.state == "pending" then
					pack.part.Transparency = 0
					pack.state = "ground"
					pack.prompt.Enabled = true
					placeOnGround(pack, pack.home)
				end
			end
		end)
	else
		Packs.dropCarried(player, false)
	end
	for _, guard in ipairs(G.World.guardians) do
		if guard.target == player then
			guard.target = nil
		end
	end
	-- de vuelta a casa
	local char = player.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if hrp and s.plot then
		hrp.CFrame = s.plot.spawnCF
	end
end

-- Compra de "Salvar el sobre"
function Packs.savePurchased(player)
	local p = pendingSave[player]
	local d = G.Data.profiles[player]
	if not d then
		return false
	end
	local rarity
	if p then
		pendingSave[player] = nil
		rarity = p.pack.rarity
		local dawn = p.pack.dawn
		destroyPack(p.pack)
		local item = { r = rarity, rem = C.RARITIES[rarity].open, dawn = dawn == true, kg = p.pack.kg }
		if #d.altarPacks < G.Core.altarCount(player) then
			table.insert(d.altarPacks, item)
		else
			table.insert(d.queue, item)
		end
	else
		-- si el tiempo se pasó, se le da un sobre de la mejor zona a la que puede entrar
		local z = 1
		for i, zone in ipairs(C.ZONES) do
			if d.speed >= zone.req then
				z = i
			end
		end
		rarity = rollRarity(z, 1)
		table.insert(d.queue, { r = rarity, rem = C.RARITIES[rarity].open, dawn = false, kg = C.rollKg() })
	end
	G.Net.toast(player, "bought")
	return true
end

---------------------------------------------------------------------
-- Altares: cuenta atrás y apertura
---------------------------------------------------------------------
function Packs.tickAltars(player, mult)
	local d = G.Data.profiles[player]
	local s = G.Core.sessions[player]
	if not d or not s then
		return
	end
	local maxA = G.Core.altarCount(player)
	while #d.altarPacks < maxA and #d.queue > 0 do
		table.insert(d.altarPacks, table.remove(d.queue, 1))
	end
	for _, p in ipairs(d.altarPacks) do
		if p.rem > 0 then
			p.rem = math.max(0, p.rem - mult)
		end
	end
	if not s.plot then
		return
	end
	for i, a in ipairs(s.plot.altars) do
		local p = d.altarPacks[i]
		if i > maxA then
			a.prompt.Enabled = false
			a.glow.Transparency = 1
			a.gui.Label.Text = ""
		elseif p then
			local rarity = C.RARITIES[p.r]
			a.glow.Color = rarity.color
			a.glow.Transparency = p.rem <= 0 and 0 or 0.5
			a.gui.Label.TextColor3 = rarity.color
			a.gui:SetAttribute("Rarity", p.r)
			a.gui:SetAttribute("Rem", p.rem)
			a.gui:SetAttribute("Kg", p.kg or 100)
			a.prompt.Enabled = p.rem <= 0
		else
			a.glow.Transparency = 1
			a.prompt.Enabled = false
			a.gui:SetAttribute("Rarity", 0)
			a.gui:SetAttribute("Rem", 0)
		end
	end
end

local function rollGod(r, dawn)
	local pool = {}
	for _, g in ipairs(C.GODS_BY_RARITY[r] or {}) do
		if not g.dawnOnly or dawn then
			table.insert(pool, g)
		end
	end
	if #pool == 0 then
		pool = C.GODS_BY_RARITY[r]
	end
	return pool[math.random(1, #pool)]
end

function Packs.openNow(player, minR, maxR)
	local d = G.Data.profiles[player]
	if not d then
		return false
	end
	for _, p in ipairs(d.altarPacks) do
		if p.rem > 0 and p.r >= minR and p.r <= maxR then
			p.rem = 0
			return true
		end
	end
	-- si no hay ninguno de ese grupo, se aplica al que más tiempo le quede
	local best
	for _, p in ipairs(d.altarPacks) do
		if p.rem > 0 and (not best or p.rem > best.rem) then
			best = p
		end
	end
	if best then
		best.rem = 0
	end
	return true
end

local function guardianStep(guard, dt)
	local model = guard.model
	local pos = model:GetPivot().Position
	local target = guard.target
	local s = target and G.Core.sessions[target]
	local hrp = target and target.Character and target.Character:FindFirstChild("HumanoidRootPart")
	if target and (not s or not s.carrying or not hrp or hrp.Position.Z < G.World.SAFE_Z) then
		guard.target = nil
		target = nil
	end
	local goal, speed
	if target and hrp then
		local d = G.Data.profiles[target]
		local hum = target.Character:FindFirstChildOfClass("Humanoid")
		local ws = hum and hum.WalkSpeed or 16
		local guardSpeed = C.ZONES[guard.zone].guardian.speed
		if d and d.speed >= guardSpeed then
			speed = math.max(18, ws * 0.82) -- eres más rápido: escapas
		else
			speed = ws * 1.18 + 4 -- te alcanza
		end
		goal = Vector3.new(hrp.Position.X, guard.height, hrp.Position.Z)
		if (Vector3.new(pos.X, 0, pos.Z) - Vector3.new(hrp.Position.X, 0, hrp.Position.Z)).Magnitude < 7 + guard.height * 0.3 then
			caught(target, guard.zone)
			return
		end
	else
		goal = guard.home
		speed = 40
	end
	local flat = Vector3.new(goal.X - pos.X, 0, goal.Z - pos.Z)
	local dist = flat.Magnitude
	local isAvatar = model:GetAttribute("Avatar")
	if dist < 0.5 then
		if isAvatar then
			G.Avatars.play(model, "idle")
		end
		return
	end
	if isAvatar then
		G.Avatars.play(model, target and "run" or "walk", target and 1.4 or 1)
	end
	local step = math.min(dist, speed * dt)
	local newPos = Vector3.new(pos.X, guard.height, pos.Z) + flat.Unit * step
	local bob = target and math.abs(math.sin(os.clock() * 10)) * 0.8 or 0
	if model:GetAttribute("Animated") or isAvatar then
		bob = 0
	end
	model:PivotTo(CFrame.lookAt(newPos + Vector3.new(0, bob, 0), newPos + flat.Unit + Vector3.new(0, bob, 0)))
	guard.phase = (guard.phase or 0) + dt * (target and 11 or 6)
	G.GodModel.animate(model, guard.phase, target and 0.9 or 0.45)
end

function Packs.start()
	RunService.Heartbeat:Connect(function(dt)
		for _, guard in ipairs(G.World.guardians) do
			guardianStep(guard, dt)
			guard.bb = guard.bb or guard.model:FindFirstChildWhichIsA("BillboardGui", true)
			local bb = guard.bb
			if bb then
				bb:SetAttribute("Chasing", guard.target ~= nil)
			end
		end
		for _, pack in pairs(Packs.list) do
			if pack.state == "carried" and pack.carrier then
				local char = pack.carrier.Character
				local head = char and char:FindFirstChild("Head")
				if head then
					pack.part.CFrame = head.CFrame * CFrame.new(0, 3, 0) * CFrame.Angles(0, math.rad(os.clock() * 120 % 360), 0)
					if pack.beam then
						pack.beam.CFrame = CFrame.new(head.Position + Vector3.new(0, 152, 0)) * CFrame.Angles(0, 0, math.rad(90))
					end
				end
				-- ¿ha llegado a su parcela?
				local s = G.Core.sessions[pack.carrier]
				if s and s.plot and char and char:FindFirstChild("HumanoidRootPart") then
					local hp = char.HumanoidRootPart.Position
					if G.World.inPlot(s.plot, hp) then
						deliver(pack.carrier, pack)
					end
				end
			elseif pack.state == "ground" or pack.state == "loose" then
				placeOnGround(pack, pack.state == "loose" and pack.loosePos or pack.home)
			end
		end
	end)
	-- control del pasillo (5 veces por segundo)
	task.spawn(function()
		while true do
			task.wait(0.2)
			local night = G.DayNight.isNight()
			for player, s in pairs(G.Core.sessions) do
				local d = G.Data.profiles[player]
				local char = player.Character
				local hrp = char and char:FindFirstChild("HumanoidRootPart")
				if d and hrp and s.plot then
					local p = hrp.Position
					if p.Z >= C.CORRIDOR_START - 3 then
						if night then
							hrp.CFrame = s.plot.spawnCF
							G.Net.toast(player, "nightClosed")
						else
							local zi = G.World.zoneAt(p.Z)
							local zone = C.ZONES[zi]
							if zi > 1 and d.speed < zone.req then
								local back = G.World.zoneStart(zi) - 10
								hrp.CFrame = CFrame.new(Vector3.new(p.X, p.Y, back), Vector3.new(p.X, p.Y, back - 10))
								G.Net.toast(player, "needSpeed", G.Format.num(zone.req), { zone = zi })
							end
						end
					end
				end
			end
		end
	end)
end

local Combat = {}

local function buildTool(weaponKey)
	local w = C.WEAPONS[weaponKey] or C.WEAPONS.bat
	local tool = Instance.new("Tool")
	tool.Name = w.es
	tool:SetAttribute("Weapon", weaponKey)
	tool.RequiresHandle = true
	tool.CanBeDropped = false
	tool.ToolTip = w.en
	tool.Grip = CFrame.new(0, -0.9, 0) * CFrame.Angles(math.rad(-35), 0, 0)
	local handle = Instance.new("Part")
	handle.Name = "Handle"
	handle.Size = Vector3.new(0.45, 2.4 + w.power * 0.1, 0.45)
	handle.Color = w.color
	handle.Material = (w.order >= 5) and Enum.Material.Neon or Enum.Material.Wood
	handle.CanCollide = false
	handle.Massless = true
	handle.Parent = tool
	local headPart = Instance.new("Part")
	headPart.Name = "Head"
	headPart.Size = weaponKey == "mjolnir" and Vector3.new(1.8, 1.1, 1.1) or Vector3.new(0.75, 1.1, 0.75)
	headPart.Color = w.color
	headPart.Material = (w.order >= 3) and Enum.Material.Metal or Enum.Material.Wood
	headPart.CanCollide = false
	headPart.Massless = true
	headPart.CFrame = handle.CFrame * CFrame.new(0, handle.Size.Y / 2 + headPart.Size.Y / 2 - 0.2, 0)
	local weld = Instance.new("WeldConstraint")
	weld.Part0 = handle
	weld.Part1 = headPart
	weld.Parent = headPart
	headPart.Parent = tool
	if w.order >= 4 then
		local pe = Instance.new("ParticleEmitter")
		pe.Rate = 10
		pe.Lifetime = NumberRange.new(0.4, 0.8)
		pe.LightEmission = 1
		pe.Color = ColorSequence.new(w.color)
		pe.Size = NumberSequence.new(0.4, 0)
		pe.Parent = headPart
	end
	return tool
end

function Combat.giveWeapon(player)
	local d = G.Data.profiles[player]
	if not d then
		return
	end
	local function clear(container)
		if container then
			for _, t in ipairs(container:GetChildren()) do
				if t:IsA("Tool") and t:GetAttribute("Weapon") then
					t:Destroy()
				end
			end
		end
	end
	clear(player:FindFirstChild("Backpack"))
	clear(player.Character)
	local tool = buildTool(d.weapon)
	tool.Activated:Connect(function()
		Combat.swing(player, tool)
	end)
	tool.Parent = player:FindFirstChild("Backpack")
end

local function lightning(a, b)
	local dist = (b - a).Magnitude
	local p = Instance.new("Part")
	p.Anchored = true
	p.CanCollide = false
	p.Material = Enum.Material.Neon
	p.Color = Color3.fromRGB(150, 210, 255)
	p.Size = Vector3.new(0.4, 0.4, dist)
	p.CFrame = CFrame.lookAt((a + b) / 2, b)
	p.Parent = workspace
	game:GetService("Debris"):AddItem(p, 0.2)
end

function Combat.swing(player, tool)
	local s = G.Core.sessions[player]
	local d = G.Data.profiles[player]
	if not s or not d then
		return
	end
	local now = os.clock()
	if now - s.lastHit < C.HIT_COOLDOWN then
		return
	end
	s.lastHit = now
	-- animación de golpe del script Animate por defecto
	local anim = Instance.new("StringValue")
	anim.Name = "toolanim"
	anim.Value = "Slash"
	anim.Parent = tool
	game:GetService("Debris"):AddItem(anim, 1)

	local char = player.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not hrp or not G.World.inCorridor(hrp.Position) then
		return
	end
	local w = C.WEAPONS[d.weapon] or C.WEAPONS.bat
	local best, bestDist = nil, C.HIT_RANGE
	for other, os2 in pairs(G.Core.sessions) do
		if other ~= player then
			local oc = other.Character
			local ohrp = oc and oc:FindFirstChild("HumanoidRootPart")
			if ohrp and G.World.inCorridor(ohrp.Position) then
				local delta = ohrp.Position - hrp.Position
				local dist = delta.Magnitude
				local facing = dist > 0.1 and hrp.CFrame.LookVector:Dot(delta.Unit) or 1
				if dist < bestDist and facing > 0.2 then
					best, bestDist = other, dist
				end
			end
		end
	end
	if not best then
		return
	end
	local vs = G.Core.sessions[best]
	local vd = G.Data.profiles[best]
	if not vs or not vd then
		return
	end
	if now < vs.invulnUntil or now < vs.shieldUntil or vs.afk then
		return
	end
	if vd.playtime < C.NEWBIE_PROTECTION then
		G.Net.toast(player, "newbie")
		return
	end
	vs.invulnUntil = now + C.HIT_INVULN
	local ohrp = best.Character.HumanoidRootPart
	local dir = Vector3.new(ohrp.Position.X - hrp.Position.X, 0, ohrp.Position.Z - hrp.Position.Z)
	dir = dir.Magnitude > 0.1 and dir.Unit or hrp.CFrame.LookVector
	local vel = dir * (70 * w.power) + Vector3.new(0, 30 + 10 * w.power, 0)
	if vel.Magnitude > 450 then
		vel = vel.Unit * 450
	end
	G.Net.Knock:FireClient(best, vel)
	if w.lightning then
		lightning(hrp.Position, ohrp.Position)
	end
	if vs.carrying then
		local rarity = vs.carrying.rarity
		G.Packs.dropCarried(best, true, ohrp.Position)
		if rarity >= C.ANNOUNCE_CHAT_FROM then
			G.Net.announce("all", "batSteal", { player.DisplayName, { rarity = rarity }, best.DisplayName }, { color = "#FF8A3D" })
		end
	end
end

	G = g
	C = g.Config
end

return Combat

-- Ciclo: 5 min de día, 15 s de noche. Al amanecer se renuevan los sobres.
local Lighting = game:GetService("Lighting")
local TweenService = game:GetService("TweenService")

local DayNight = {}
local G
local C

local night = false
local endsAt = 0
local warned = false

function DayNight.isNight()
	return night
end

function DayNight.timeLeft()
	return math.max(0, endsAt - os.clock())
end

local function tweenLight(toNight)
	local info = TweenInfo.new(2.5, Enum.EasingStyle.Sine)
	TweenService:Create(Lighting, info, {
		ClockTime = toNight and 0.5 or 16.3,
		Brightness = toNight and 0.7 or 3,
		OutdoorAmbient = toNight and Color3.fromRGB(50, 45, 100) or Color3.fromRGB(175, 145, 125),
	}):Play()
	local grade = Lighting:FindFirstChild("Grade")
	if grade then
		TweenService:Create(grade, info, { TintColor = toNight and Color3.fromRGB(170, 170, 255) or Color3.fromRGB(255, 240, 222) }):Play()
	end
	local gate = G.World.nightGate
	if gate then
		TweenService:Create(gate, info, { Transparency = toNight and 0.15 or 1 }):Play()
		gate.CanCollide = toNight
	end
end

local function startNight()
	night = true
	endsAt = os.clock() + C.NIGHT_SECONDS
	warned = false
	tweenLight(true)
	G.Packs.nightfall()
	G.Net.announce("all", "nightStart", {}, { color = "#A78BFA", big = true })
end

local function startDay()
	night = false
	endsAt = os.clock() + C.DAY_SECONDS
	warned = false
	tweenLight(false)
	G.Packs.refreshAll()
end

-- Producto "Noche para el servidor": adelanta la noche
function DayNight.forceNight(buyerName)
	if not night then
		endsAt = math.min(endsAt, os.clock() + 3)
		G.Net.announce("all", "nightBought", { buyerName }, { color = "#A78BFA", big = true })
		warned = true
	end
end

function DayNight.init(g)
	G = g
	C = g.Config
end

function DayNight.start()
	startDay()
	task.spawn(function()
		while true do
			task.wait(0.25)
			local left = endsAt - os.clock()
			if not night and not warned and left <= C.NIGHT_WARNING then
				warned = true
				G.Net.announce("all", "nightSoon", { math.ceil(left) }, { color = "#A78BFA" })
			end
			if left <= 0 then
				if night then
					startDay()
				else
					startNight()
				end
			end
		end
	end)
end

return DayNight

local DataStoreService = game:GetService("DataStoreService")

local Boards = {}
local G

local stores = {}
local names = {}
local lastTop10 = { speed = {}, money = {}, time = {} }

-- Las cifras absurdas no caben en un entero: se guardan en escala logarítmica
local function encode(v)
	return math.floor(math.log10(math.max(1, v)) * 1e9)
end
local function decode(x)
	return 10 ^ (x / 1e9)
end

local function nameOf(userId)
	if names[userId] then
		return names[userId]
	end
	local ok, n = pcall(function()
		return Players:GetNameFromUserIdAsync(userId)
	end)
	names[userId] = ok and n or ("#" .. userId)
	return names[userId]
end

local COLORS = {
	Color3.fromRGB(255, 215, 0),
	Color3.fromRGB(200, 205, 215),
	Color3.fromRGB(205, 127, 50),
}

local function render(board, rows, fmt)
	for _, c in ipairs(board.list:GetChildren()) do
		if c:IsA("TextLabel") then
			c:Destroy()
		end
	end
	for i, row in ipairs(rows) do
		local l = Instance.new("TextLabel")
		l.LayoutOrder = i
		l.Size = UDim2.new(1, 0, 0, 34)
		l.BackgroundTransparency = i % 2 == 0 and 1 or 0.85
		l.BackgroundColor3 = Color3.new(1, 1, 1)
		l.Font = Enum.Font.GothamBold
		l.TextSize = 24
		l.TextXAlignment = Enum.TextXAlignment.Left
		l.TextColor3 = COLORS[i] or Color3.new(1, 1, 1)
		l.Text = string.format("  #%d  %s  —  %s", i, row.name, fmt(row.value))
		l.Parent = board.list
	end
end

local function update()
	-- 1) subir los datos de los jugadores de este servidor
	for player in pairs(G.Core.sessions) do
		local d = G.Data.profiles[player]
		if d then
			local key = tostring(player.UserId)
			pcall(function()
				stores.speed:SetAsync(key, encode(d.bestSpeed))
			end)
			pcall(function()
				stores.money:SetAsync(key, encode(d.totalMoney))
			end)
			pcall(function()
				stores.time:SetAsync(key, math.floor(d.playtime))
			end)
		end
	end
	-- 2) leer el top 50 de cada uno
	local defs = {
		{ id = "speed", fmt = function(x)
			return "⚡ " .. G.Format.num(decode(x))
		end },
		{ id = "money", fmt = function(x)
			return "💰 " .. G.Format.num(decode(x))
		end },
		{ id = "time", fmt = function(x)
			return "⏱️ " .. G.Format.time(x)
		end },
	}
	for _, def in ipairs(defs) do
		local ok, pages = pcall(function()
			return stores[def.id]:GetSortedAsync(false, 50)
		end)
		if ok and pages then
			local rows = {}
			local top10 = {}
			for _, entry in ipairs(pages:GetCurrentPage()) do
				local uid = tonumber(entry.key)
				table.insert(rows, { name = uid and nameOf(uid) or entry.key, value = entry.value })
				if #rows <= 10 and uid then
					top10[uid] = true
				end
			end
			render(G.World.boards[def.id], rows, def.fmt)
			-- aviso al entrar en el top 10
			for player in pairs(G.Core.sessions) do
				local uid = player.UserId
				if top10[uid] and not lastTop10[def.id][uid] and next(lastTop10[def.id]) ~= nil then
					G.Net.announce("all", "top10", { player.DisplayName, { stat = def.id } }, { color = "#FFD23F", big = true })
				end
			end
			lastTop10[def.id] = top10
		end
	end
end

local function rcolor(r)
	local rr = Config.RARITIES[r]
	if not rr then
		return WHITE
	end
	return rr.id == 10 and Color3.fromRGB(200, 120, 255) or rr.color
end

local function new(class, props, children)
	local o = Instance.new(class)
	for k, v in pairs(props or {}) do
		if k ~= "Parent" then
			o[k] = v
		end
	end
	for _, c in ipairs(children or {}) do
		c.Parent = o
	end
	if props and props.Parent then
		o.Parent = props.Parent
	end
	return o
end

local function corner(r)
	return new("UICorner", { CornerRadius = UDim.new(0, r or 14) })
end

local function stroke(color, th, tr)
	return new("UIStroke", { Color = color or GOLD, Thickness = th or 2, Transparency = tr or 0, ApplyStrokeMode = Enum.ApplyStrokeMode.Border })
end

local function textStroke(color, th)
	return new("UIStroke", { Color = color or Color3.fromRGB(10, 6, 24), Thickness = th or 2.5, ApplyStrokeMode = Enum.ApplyStrokeMode.Contextual })
end

local function gradient(c1, c2, rot)
	return new("UIGradient", { Color = ColorSequence.new(c1, c2), Rotation = rot or 90 })
end

local function darker(c, k)
	k = k or 0.55
	return Color3.new(c.R * k, c.G * k, c.B * k)
end

local function opt(key)
	local st = state.settings
	if not st then
		return key ~= "ui" and true or 2
	end
	if key == "ui" then
		return st.ui or 2
	end
	return st[key] ~= false
end

local function play(name, pitch)
	if not opt("sfx") then
		return
	end
	local s = SFX[name]
	if s then
		s.PlaybackSpeed = pitch or s.PlaybackSpeed
		s:Play()
	end
end

-- Animación de "rebote" al tocar
local function pressable(gui)
	local scale = gui:FindFirstChildOfClass("UIScale") or new("UIScale", { Parent = gui })
	local function to(v, style)
		TweenService:Create(scale, TweenInfo.new(0.12, style or Enum.EasingStyle.Quad), { Scale = v }):Play()
	end
	gui.MouseEnter:Connect(function()
		to(1.05)
	end)
	gui.MouseLeave:Connect(function()
		to(1)
	end)
	gui.MouseButton1Down:Connect(function()
		to(0.9)
	end)
	gui.MouseButton1Up:Connect(function()
		TweenService:Create(scale, TweenInfo.new(0.3, Enum.EasingStyle.Back), { Scale = 1 }):Play()
	end)
end

local function resolveArg(a)
	if type(a) == "table" then
		if a.rarity then
			return string.upper(Locale.name(Config.RARITIES[a.rarity]))
		elseif a.zone then
			return Locale.name(Config.ZONES[a.zone])
		elseif a.guardian then
			return Locale.name(Config.ZONES[a.guardian].guardian)
		elseif a.god then
			local g = Config.GOD_BY_ID[a.god]
			return g and g.name or "?"
		elseif a.myth then
			return Locale.name(Config.MYTHS[a.myth])
		elseif a.weapon then
			return Locale.name(Config.WEAPONS[a.weapon])
		elseif a.stat then
			return t(a.stat == "speed" and "speed" or (a.stat == "money" and "money" or "time"))
		end
		return "?"
	end
	return tostring(a)
end

local function fmtKey(key, args)
	local r = {}
	for i, a in ipairs(args or {}) do
		r[i] = resolveArg(a)
	end
	return t(key, table.unpack(r))
end

-- Vista 3D de un dios
local function godViewport(parent, god, size, pos, z)
	local vp = new("ViewportFrame", {
		BackgroundTransparency = 1,
		Size = size,
		Position = pos or UDim2.new(),
		Ambient = Color3.fromRGB(170, 160, 190),
		LightColor = Color3.fromRGB(255, 245, 230),
		LightDirection = Vector3.new(-0.4, -1, -0.6),
		ZIndex = z or 1,
		Parent = parent,
	})
	local model = GodModel.build(god, rcolor(god.r))
	for _, d in ipairs(model:GetDescendants()) do
		if d:IsA("ParticleEmitter") or d:IsA("Fire") or d:IsA("PointLight") then
			d:Destroy()
		end
	end
	model.Parent = vp
	local cam = new("Camera", { FieldOfView = 32, Parent = vp })
	vp.CurrentCamera = cam
	local h = math.max(4, model:GetExtentsSize().Y + 0.5)
	cam.CFrame = CFrame.lookAt(Vector3.new(0, h * 0.55, -h * 2.2), Vector3.new(0, h * 0.5, 0))
	return vp, model
end

local function render(force)
	if not panel.Visible or not currentPanel then
		return
	end
	local sigFn = signatures[currentPanel]
	local sig = sigFn and sigFn(currentTab) or tostring(os.clock())
	if not force and content:GetAttribute("Sig") == sig then
		return
	end
	content:SetAttribute("Sig", sig)
	local pos = content.CanvasPosition
	clearContent()
	renderers[currentPanel](currentTab)
	content.CanvasPosition = pos
end

local tabList = {}
local function setTabs(list)
	tabList = list
	for _, c in ipairs(tabs:GetChildren()) do
		if c:IsA("TextButton") then
			c:Destroy()
		end
	end
	for i, tb in ipairs(list) do

		button({ Parent = tabs, Text = tb.text, Size = UDim2.new(0, 180, 1, 0), LayoutOrder = i, ZIndex = 12, Color = on and GOLD or Color3.fromRGB(80, 60, 140), OnClick = function()

			setTabs(list)
			content.CanvasPosition = Vector2.zero
			render(true)
		end })
	end
end

local function shake(power, duration)
	if not opt("shake") then
		return
	end
	local cam = workspace.CurrentCamera
	local t0 = os.clock()
	local conn
	conn = RunService.RenderStepped:Connect(function()
		local e = os.clock() - t0
		if e > duration then
			conn:Disconnect()
			return
		end
		local k = power * (1 - e / duration)
		cam.CFrame = cam.CFrame * CFrame.new((math.random() - 0.5) * k, (math.random() - 0.5) * k, 0)
	end)
end

local function burst(color, n)
	for k = 1, n do
		local sz = math.random(10, 22)
		local p = new("Frame", { BackgroundColor3 = k % 3 == 0 and WHITE or color, Size = UDim2.fromOffset(sz, sz), Position = UDim2.fromScale(0.5, 0.5), AnchorPoint = Vector2.new(0.5, 0.5), Rotation = 45, ZIndex = 39, Parent = reveal }, { corner(4) })
		local a = math.random() * math.pi * 2
		local d = math.random(220, 620)
		TweenService:Create(p, TweenInfo.new(0.9 + math.random() * 0.5, Enum.EasingStyle.Quint), { Position = UDim2.new(0.5, math.cos(a) * d, 0.5, math.sin(a) * d), Rotation = math.random(-360, 360), BackgroundTransparency = 1 }):Play()
		task.delay(1.5, function()
			p:Destroy()
		end)
	end
end

local function doReveal(data)
	local g = Config.GOD_BY_ID[data.god]
	if not g then
		return
	end
	revealing = true
	local rarity = Config.RARITIES[g.r]
	local rc = rcolor(g.r)
	-- 1) oscurecer
	reveal.Visible = true
	reveal.BackgroundTransparency = 1
	godCard.Visible = false
	packCard.Visible = true
	rays.Visible = true
	for _, r in ipairs(rayFrames) do
		r.BackgroundColor3 = rc
	end
	TweenService:Create(reveal, TweenInfo.new(0.4), { BackgroundTransparency = 0.15 }):Play()
	packGrad.Color = ColorSequence.new(rc, darker(rc, 0.35))
	packStroke.Color = WHITE
	packLabel.Text = string.upper(Locale.name(rarity))
	packLabel.TextColor3 = WHITE
	packScale.Scale = 0
	TweenService:Create(packScale, TweenInfo.new(0.45, Enum.EasingStyle.Back), { Scale = 1 }):Play()
	play("pop")
	-- 2) carga con temblor que aumenta
	local charge = 1.1 + math.min(g.r, 10) * 0.14
	local t0 = os.clock()
	local conn
	conn = RunService.RenderStepped:Connect(function(dt)
		local e = os.clock() - t0
		local k = math.clamp(e / charge, 0, 1)
		rays.Rotation += dt * (30 + k * 260)
		packCard.Rotation = (math.random() - 0.5) * 14 * k
		packCard.Position = UDim2.new(0.5, (math.random() - 0.5) * 18 * k, 0.5, (math.random() - 0.5) * 18 * k)
		packScale.Scale = 1 + 0.12 * k + math.sin(e * 18) * 0.03 * k
		if rarity.rainbow or rarity.sparkle then
			local hue = (os.clock() * 0.5) % 1
			packStroke.Color = Color3.fromHSV(hue, 0.7, 1)
		end
		if e >= charge then
			conn:Disconnect()
		end
	end)
	task.wait(charge)
	-- 3) destello + explosión
	packCard.Visible = false
	packCard.Rotation = 0
	flash.BackgroundTransparency = 0
	TweenService:Create(flash, TweenInfo.new(0.5), { BackgroundTransparency = 1 }):Play()
	burst(rc, 26 + g.r * 4)
	for k = 0, 2 do
		local ring = new("Frame", { BackgroundTransparency = 1, Size = UDim2.fromOffset(40, 40), Position = UDim2.fromScale(0.5, 0.5), AnchorPoint = Vector2.new(0.5, 0.5), ZIndex = 38, Parent = reveal }, { corner(999), stroke(k == 1 and WHITE or rc, 8 - k * 2) })
		local st = ring:FindFirstChildOfClass("UIStroke")
		TweenService:Create(ring, TweenInfo.new(0.7 + k * 0.2, Enum.EasingStyle.Quart), { Size = UDim2.fromOffset(900 + k * 200, 900 + k * 200) }):Play()
		TweenService:Create(st, TweenInfo.new(0.7 + k * 0.2), { Transparency = 1 }):Play()
		task.delay(1.2, function()
			ring:Destroy()
		end)
	end
	play("whoosh", 1.2)
	if g.r >= Config.ANNOUNCE_BIG_FROM then
		shake(1.4, 1.4)
	elseif g.r >= 5 then
		shake(0.5, 0.5)
	end
	-- 4) carta con el dios en 3D
	for _, c in ipairs(godVPHolder:GetChildren()) do
		c:Destroy()
	end
	local _, model = godViewport(godVPHolder, g, UDim2.fromScale(1, 1), nil, 36)
	spinModel = model
	godGrad.Color = ColorSequence.new(darker(rc, 0.75), Color3.fromRGB(16, 10, 34))
	godStroke.Color = rc
	godName.Text = g.name
	godNameGrad.Color = ColorSequence.new(WHITE, rc)
	ribbon.BackgroundColor3 = rc
	ribbonLabel.Text = string.upper(Locale.name(rarity))
	ribbonLabel.TextColor3 = (g.r == 7 or g.r == 12) and DARK or WHITE
	local kg = data.kg or 100
	godIncome.Text = string.format("⚖️ %d kg   💰 +%s/s", kg, Format.num(rarity.income * (kg / 100) * (state.moneyMult or 1)))
	godFact.Text = Locale.lang() == "es" and g.factEs or g.factEn
	godNew.Text = data.isNew and t("gotNew") or ""
	godCard.Visible = true
	godScale.Scale = 0.2
	TweenService:Create(godScale, TweenInfo.new(0.55, Enum.EasingStyle.Back), { Scale = 1 }):Play()
	play("reward", 0.7)
	if data.isNew then
		bump(godNew, 1.6)
	end
	task.wait(0.4)
	revealing = false
end

local function locLabel(bb)
	local l = bb:FindFirstChild("Label")
	if not l then
		return
	end
	l.Font = TITLE
	if bb:GetAttribute("Overhead") then
		l.FontFace = HEAVY
		l.Text = string.format('<font color="#FFB347">[%s]</font>\n<font color="#7CFF8A">$%s/s</font>', t("title" .. (bb:GetAttribute("Title") or 1)), Format.num(bb:GetAttribute("Income") or 0))
		return
	end
	if bb:GetAttribute("Altar") then
		local r = bb:GetAttribute("Rarity") or 0
		if r > 0 then
			local rem = bb:GetAttribute("Rem") or 0
			l.TextColor3 = rcolor(r)
			l.Text = string.format("%s · %d kg\n%s", Locale.name(Config.RARITIES[r]), bb:GetAttribute("Kg") or 100, rem <= 0 and ("✨ " .. t("ready")) or ("⏳ " .. Format.time(rem)))
		else
			l.Text = ""
		end

		local zone = Config.ZONES[bb:GetAttribute("Guardian")]
		local chasing = bb:GetAttribute("Chasing")
		l.Text = string.format("%s%s", Locale.name(zone.guardian), chasing and ("\n⚠️ " .. t("chasing")) or "")

		local zi = bb:GetAttribute("Zone")

local walkers = {}
local ANIM = { idle = "rbxassetid://507766666", walk = "rbxassetid://507777826" }
local animSets = setmetatable({}, { __mode = "k" })
local function playAnim(model, name, speed)
	local hum = model:FindFirstChildOfClass("Humanoid")
	local animator = hum and hum:FindFirstChildOfClass("Animator")
	if not animator then
		return
	end
	local set = animSets[model]
	if not set then
		set = {}
		animSets[model] = set
	end
	if set.current == name then
		return
	end
	if set.current and set[set.current] then
		set[set.current]:Stop(0.25)
	end
	if not set[name] then
		local a = Instance.new("Animation")
		a.AnimationId = ANIM[name]
		local ok, track = pcall(function()
			return animator:LoadAnimation(a)
		end)
		if not ok then
			return
		end
		track.Looped = true
		set[name] = track
	end
	set[name]:Play(0.25)
	if speed then
		set[name]:AdjustSpeed(speed)
	end
	set.current = name
end

end

	return Vector3.new(-19 + math.random() * 38, 0, -17 + math.random() * 30)
end
local walkAcc = 0
RunService.Heartbeat:Connect(function(dt)
	walkAcc += dt
	if walkAcc < 1 / 30 then
		return
	end
	dt = walkAcc
	walkAcc = 0
	local cam = workspace.CurrentCamera
	local now = os.clock()
	for _, folder in ipairs(plotFolders()) do
		local cf = folder:GetAttribute("PlotCF")
		if cam and (cam.CFrame.Position - cf.Position).Magnitude < 220 then
			for _, m in ipairs(folder:GetChildren()) do
				if m:IsA("Model") and m:GetAttribute("Income") and m.PrimaryPart then
					local st = walkers[m]
					if not st then
						local rel = cf:ToObjectSpace(m:GetPivot())
						st = { pos = Vector3.new(rel.X, 0, rel.Z), y = rel.Y, target = newTarget(), wait = math.random() * 2, yaw = math.pi, speed = 3 + math.random() * 2.5, phase = math.random() * 6 }
						walkers[m] = st
						m.AncestryChanged:Connect(function()
							if not m.Parent then
								walkers[m] = nil
							end
						end)
					end
					local moving = false
					if st.wait > 0 then
						st.wait -= dt
					else
						local delta = st.target - st.pos
						local dist = delta.Magnitude
						if dist < 0.5 then
							st.target = newTarget()
							st.wait = 1 + math.random() * 3
						else
							moving = true
							st.pos += delta.Unit * math.min(dist, st.speed * dt)
							local goalYaw = math.atan2(-delta.X, -delta.Z)
							local diff = (goalYaw - st.yaw + math.pi) % (2 * math.pi) - math.pi
							st.yaw += diff * math.min(1, dt * 6)
						end
					end
					local bob = moving and math.abs(math.sin(now * 9 + st.phase)) * 0.35 or math.sin(now * 2 + st.phase) * 0.08
					local tilt = moving and math.sin(now * 9 + st.phase) * 0.08 or 0
					local animated = m:GetAttribute("Animated")
					if m:GetAttribute("Avatar") then
						bob, tilt = 0, 0
						playAnim(m, moving and "walk" or "idle", moving and st.speed / 5 or 1)
					end
					if animated then
						bob, tilt = moving and math.abs(math.sin(now * 9 + st.phase)) * 0.12 or 0, 0
					end
					m:PivotTo(cf * CFrame.new(st.pos.X, st.y + bob, st.pos.Z) * CFrame.Angles(0, st.yaw, tilt))
					if animated then
						GodModel.animate(m, now * 9 + st.phase, moving and 0.55 or 0)
					end
					local runes = m:FindFirstChild("Runes")
					if runes then
						st.runeY = st.runeY or (m:GetPivot():ToObjectSpace(runes.CFrame).Y - bob)
						runes.CFrame = m:GetPivot() * CFrame.new(0, st.runeY - bob, 0) * CFrame.Angles(0, now * 0.8, 0)
					end
				end
			end
		end
	end
end)

local runTrack = nil
local belts = {}
local function myTreadmill()
	local plot = myPlot()
	return plot and plot:FindFirstChild("Treadmill")
end
local function ensureBelt(tm)
	if #belts > 0 and belts[1].Parent and belts.tm == tm then
		return
	end
	for _, b in ipairs(belts) do
		b:Destroy()
	end
	belts = { tm = tm }
	for k = 1, 8 do
		local stripe = new("Part", { Anchored = true, CanCollide = false, CanQuery = false, Size = Vector3.new(tm.Size.X - 0.4, 0.05, 0.35), Color = Color3.fromRGB(90, 230, 255), Material = Enum.Material.Neon, Transparency = 0.35, Parent = workspace })
		table.insert(belts, stripe)
	end
end
local beltOffset = 0
RunService.RenderStepped:Connect(function(dt)
	local tm = myTreadmill()
	local running = state.onTreadmill == true or state.afk == true
	if tm then
		ensureBelt(tm)
		beltOffset = (beltOffset + dt * (running and (10 + (state.combo or 1) * 4) or 1.5)) % tm.Size.Z
		for k, stripe in ipairs(belts) do
			local z = ((k - 1) * tm.Size.Z / 8 + beltOffset) % tm.Size.Z - tm.Size.Z / 2
			stripe.CFrame = tm.CFrame * CFrame.new(0, tm.Size.Y / 2 + 0.03, -z)
		end
	end
	local char = player.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local animator = hum and hum:FindFirstChildOfClass("Animator")
	if running and animator then
		if not runTrack then
			local animate = char:FindFirstChild("Animate")
			local runFolder = animate and animate:FindFirstChild("run")
			local anim = runFolder and runFolder:FindFirstChildOfClass("Animation")
			if anim then
				runTrack = animator:LoadAnimation(anim)
				runTrack.Looped = true
				runTrack.Priority = Enum.AnimationPriority.Action
				runTrack:Play(0.2)
			end
		end
		if runTrack then
			runTrack:AdjustSpeed(1.2 + (state.combo or 1) * 0.35)
		end
	elseif runTrack then
		runTrack:Stop(0.2)
		runTrack = nil
	end
end)
player.CharacterAdded:Connect(function()
	runTrack = nil
end)

local lastInput = os.clock()
local rejoinSent = false
UserInputService.InputBegan:Connect(function()
	lastInput = os.clock()
end)
local function tryRejoin()
	if state.afk and not rejoinSent then
		rejoinSent = true
		Action:FireServer("rejoin")
		task.delay(30, function()
			rejoinSent = false
		end)
	end
end
player.Idled:Connect(function(idleTime)
	if idleTime >= Config.AFK_REJOIN_AFTER then
		tryRejoin()
	end
end)
task.spawn(function()
	while true do
		task.wait(30)
		if os.clock() - lastInput >= Config.AFK_REJOIN_AFTER then
			tryRejoin()
		end
	end
end)

local function add(...)
	local t = {}
	for i = 1, select("#", ...) do
		t[i] = tostring((select(i, ...)))
	end
	table.insert(out, table.concat(t, " "))
end
local function step(name, fn)
	local ok, err = pcall(fn)
	if ok then
		add("✅", name)
	else
		fails += 1
		add("❌", name, "->", err)
	end
end
